    nama TEXT NOT NULL, alamat TEXT, email TEXT,
    telepon TEXT, tanggal_daftar TEXT, saldo REAL DEFAULT 0
);
ALTER TABLE setoran ADD COLUMN anggota_id INTEGER DEFAULT NULL;
UPDATE setoran SET anggota_id = (SELECT id FROM anggota WHERE anggota.nama = setoran.nama_anggota);

## Task 14: Saldo Anggota Auto dari Total Setoran

### Konsep
Hapus kolom `saldo` dari tabel `anggota`. Saldo dihitung real-time via SQL JOIN:
```sql
SELECT a.*, COALESCE(SUM(s.saldo), 0) as saldo
FROM anggota a LEFT JOIN setoran s ON a.id = s.anggota_id
GROUP BY a.id
```
